# Written for the book as a draft to review.
from abc import ABC, abstractmethod
from decimal import ROUND_HALF_UP, Decimal

from review_gate.domain import Consignment, Money, RatingError


class RateStrategy(ABC):
    @abstractmethod
    def calculate(self, consignment: Consignment) -> Money: ...


class FlatRate(RateStrategy):
    def calculate(self, consignment: Consignment) -> Money:
        return Money(Decimal("10.00"))


class PerKilogramRate(RateStrategy):
    def __init__(self, rate: Decimal) -> None:
        self._rate = rate

    def calculate(self, consignment: Consignment) -> Money:
        amount = consignment.weight.kilograms * self._rate
        return Money(amount.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP))


class RateStrategyFactory:
    @staticmethod
    def create(name: str) -> RateStrategy:
        match name:
            case "flat":
                return FlatRate()
            case "per_kg":
                return PerKilogramRate(Decimal("0.45"))
            case _:
                raise RatingError(f"unknown rule {name}")


def price(rule: str, consignment: Consignment, /) -> Money:
    return RateStrategyFactory.create(rule).calculate(consignment)
